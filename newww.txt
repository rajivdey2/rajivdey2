import pandas as pd

df = pd.read_csv('marks.csv')

# Display first 5 rows
df.head()




df['Average'] = df[['Mathematics', 'Physics', 'Chemistry']].mean(axis=1)

df.head()




def classify_performance(avg):
    if avg < 50:
        return 'Low'
    elif avg < 75:
        return 'Medium'
    else:
        return 'High'

df['Performance'] = df['Average'].apply(classify_performance)






df.head(10)






print(df['Performance'].value_counts())






df.to_csv('student_performance_output.csv', index=False)
